// ============================================================ // نظام إدارة الرواتب — Google Apps Script Web App // انسخ هذا الكود كاملاً في محرر Apps Script الخاص بجدول بياناتك // ثم: نشر ← نشر كتطبيق ويب ← الوصول: الجميع // ============================================================ // ضع معرّف جدول البيانات هنا (من رابط الشيت) const SHEET_ID = 'YOUR_SPREADSHEET_ID_HERE'; // أسماء الأوراق — يجب أن تطابق أسماء الشيتات في جدول البيانات const SHEETS = ['employees', 'loans', 'deductions', 'arrears', 'bonuses', 'attendance', 'settings']; // ====================================================== // doPost — نقطة الدخول الرئيسية // ====================================================== function doPost(e) { try { var body = JSON.parse(e.postData.contents); var action = body.action; var payload = body.payload || {}; var ss = SpreadsheetApp.openById(SHEET_ID); // تهيئة الأوراق إن لم تكن موجودة initSheets(ss); switch (action) { // -------------------------------------------------- // getAll — تحميل جميع البيانات دفعةً واحدة // -------------------------------------------------- case 'getAll': { var result = {}; ['employees','loans','deductions','arrears','bonuses','attendance'].forEach(function(name) { var sh = ss.getSheetByName(name); var last = sh.getLastRow(); if (!sh || last < 2) { result[name] = []; return; } var rows = sh.getRange(1, 1, last, sh.getLastColumn()).getValues(); var headers = rows[0]; result[name] = rows.slice(1) .filter(function(row) { return row[0] !== '' && row[0] !== null; }) .map(function(row) { var obj = {}; headers.forEach(function(h, i) { obj[h] = (row[i] === null || row[i] === undefined) ? '' : row[i]; }); return obj; }); }); // تحميل الإعدادات var settingsSh = ss.getSheetByName('settings'); if (settingsSh && settingsSh.getLastRow() > 0) { var sRows = settingsSh.getDataRange().getValues(); var settings = {}; sRows.forEach(function(r) { if (r[0]) settings[String(r[0])] = r[1]; }); result.settings = settings; } return jsonOut({ success: true, data: result }); } // -------------------------------------------------- // syncRecord — إضافة / تعديل / حذف سجل واحد // -------------------------------------------------- case 'syncRecord': { var sheetName = payload.sheet; var record = payload.record; var operation = payload.operation; // 'upsert' | 'delete' if (!sheetName || !record) { return jsonOut({ success: false, error: 'Missing sheet or record' }); } var sh = ss.getSheetByName(sheetName); if (!sh) sh = ss.insertSheet(sheetName); var lastRow = sh.getLastRow(); var lastCol = sh.getLastColumn(); // ---- حذف ---- if (operation === 'delete') { if (lastRow < 2) return jsonOut({ success: true }); var ids = sh.getRange(2, 1, lastRow - 1, 1).getValues(); for (var r = 0; r < ids.length; r++) { if (String(ids[r][0]) === String(record.id)) { sh.deleteRow(r + 2); return jsonOut({ success: true }); } } return jsonOut({ success: true }); // لم يُعثر عليه — لا بأس } // ---- upsert ---- var headers; if (lastRow === 0 || lastCol === 0) { // الورقة فارغة — أنشئ الترويسات headers = Object.keys(record); sh.appendRow(headers); lastRow = 1; } else { headers = sh.getRange(1, 1, 1, lastCol).getValues()[0]; } var values = headers.map(function(h) { return record[h] !== undefined ? record[h] : ''; }); // هل السجل موجود بالفعل؟ if (lastRow > 1) { var existingIds = sh.getRange(2, 1, lastRow - 1, 1).getValues(); for (var r = 0; r < existingIds.length; r++) { if (String(existingIds[r][0]) === String(record.id)) { sh.getRange(r + 2, 1, 1, values.length).setValues([values]); return jsonOut({ success: true }); } } } // سجل جديد sh.appendRow(values); return jsonOut({ success: true }); } // -------------------------------------------------- // saveSettings — حفظ إعدادات النظام // -------------------------------------------------- case 'saveSettings': { var sh = ss.getSheetByName('settings'); if (!sh) sh = ss.insertSheet('settings'); sh.clearContents(); Object.keys(payload).forEach(function(key) { sh.appendRow([key, payload[key]]); }); return jsonOut({ success: true }); } default: return jsonOut({ success: false, error: 'Unknown action: ' + action }); } } catch (err) { console.error(err); return jsonOut({ success: false, error: err.toString() }); } } // ====================================================== // doGet — للتحقق من أن التطبيق يعمل (اختياري) // ====================================================== function doGet(e) { return jsonOut({ status: 'ok', message: 'نظام الرواتب — Web App يعمل بنجاح' }); } // ====================================================== // مساعد: إنشاء رد JSON // ====================================================== function jsonOut(obj) { var output = ContentService.createTextOutput(JSON.stringify(obj)); output.setMimeType(ContentService.MimeType.JSON); return output; } // ====================================================== // مساعد: تهيئة الأوراق بالترويسات المطلوبة // يُنفَّذ تلقائياً عند أول استدعاء — آمن للتكرار // ====================================================== function initSheets(ss) { var schemas = { employees: ['id','name','natId','dept','pos','basic','housing','transport','phone','other','iban','joinDate','status'], loans: ['id','empId','type','amount','remaining','monthly','date','reason','status'], deductions: ['id','empId','type','amount','month','desc','status'], arrears: ['id','empId','type','amount','months','date','desc','status'], bonuses: ['id','empId','type','amount','month','desc','status'], attendance: ['id','empId','month','present','absent','late','leaves','approved'], settings: ['key','value'] }; Object.keys(schemas).forEach(function(name) { var sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(schemas[name]); // تنسيق الترويسة sh.getRange(1, 1, 1, schemas[name].length) .setBackground('#0f2942') .setFontColor('#ffffff') .setFontWeight('bold'); } }); } // ====================================================== // دالة مساعدة: تشغيلها مرة واحدة يدوياً من المحرر // لتهيئة جدول البيانات قبل البدء // ====================================================== function setupSpreadsheet() { var ss = SpreadsheetApp.openById(SHEET_ID); initSheets(ss); // إضافة إعدادات افتراضية var settingsSh = ss.getSheetByName('settings'); if (settingsSh.getLastRow() < 2) { var defaults = [ ['company', 'نظام الرواتب'], ['currency', 'ريال'], ['gosi', 9], ['workDays', 30] ]; defaults.forEach(function(row) { settingsSh.appendRow(row); }); } SpreadsheetApp.getUi().alert('✅ تم إعداد جدول البيانات بنجاح!'); }
